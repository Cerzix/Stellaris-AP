namespace = ap

# Game start / save load: find the human player and set up the bridge
event = {
	id = ap.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_playable_country = {
			limit = { is_ai = no }
			ap_setup_player = yes
		}
		# Countries with this flag research the normal technologies, see NOT_AI in build_mod.py
		every_country = {
			limit = { is_ai = yes }
			set_country_flag = ap_ai
		}
	}
}

# Deaths in the player's empire. The client reads them from the log as "AP_DEATH <kind> <name>", see DEATHS
# there. Not while a doom is carried out: that is a death that was received, not one to send.

# A leader died. This = country, From = leader
country_event = {
	id = ap.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = ap_player
		NOT = { has_country_flag = ap_dooming }
		NOT = { has_country_flag = ap_starting }
	}

	immediate = {
		if = {
			limit = { from = { is_ruler = yes } }
			log = "AP_DEATH 1 [From.GetName]"
		}
		else = {
			log = "AP_DEATH 2 [From.GetName]"
		}
	}
}

# A fleet lost a battle, or was destroyed. This = its owner, FromFrom = the fleet. A lost battle is reported
# whether the fleet got away or not; a fleet that is gone by then is reported as destroyed instead. The
# client only sends one death in a while, so a battle of many fleets is one death
country_event = {
	id = ap.31
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = ap_player
		NOT = { has_country_flag = ap_dooming }
		exists = fromfrom
		fromfrom = { is_ship_class = shipclass_military }
	}

	immediate = {
		log = "AP_DEATH 3 [FromFrom.GetName]"
	}
}

# A colony was invaded. This = planet, From = its owner
planet_event = {

	}

	immediate = {
		log = "AP_DEATH 4 [This.GetName]"
	}
}

# A colony was destroyed. This = planet, which still has its owner
planet_event = {
	id = ap.33
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		owner = {
			has_country_flag = ap_player
			NOT = { has_country_flag = ap_dooming }
		}
	}

	immediate = {
		log = "AP_DEATH 5 [This.GetName]"
	}
}
